
# Program: N-Queens Using Backtracking

N = 4

# Create chessboard
board = [-1] * N


# Function to check safe position
def is_safe(row, col):

    for i in range(row):

        # Check same column
        if board[i] == col:
            return False

        # Check diagonal
        if abs(board[i] - col) == abs(i - row):
            return False

    return True


# Function to solve N-Queens
def solve_n_queens(row):

    # All queens are placed
    if row == N:
        return True

    # Try every column
    for col in range(N):

        if is_safe(row, col):

            # Place queen
            board[row] = col

            # Move to next row
            if solve_n_queens(row + 1):
                return True

            # Backtracking
            board[row] = -1

    return False


# Main program
if solve_n_queens(0):

    print("N-Queens Solution:")
    print()

    for i in range(N):

        for j in range(N):

            if board[i] == j:
                print("Q", end=" ")

            else:
                print("-", end=" ")

        print()

else:
    print("Solution Not Found")
